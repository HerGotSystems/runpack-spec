# RunPack Example — Basic task (block format)
# Minimal task packet. No session. No retry. Plain text output.
# Works with: ChatGPT, Claude, Gemini, any local LLM.

=== RUNPACK_TASK ===
version: 1.1
legacy_aip_version: 1.1
aip_version: 1.1
packet_type: runpack_task
task_id: sum-001
task:    document_summary

input:
  text: [USER_INPUT]
  Paste or insert document content here.
  [/USER_INPUT]

rules:
  3_sentences_maximum
  plain_language
  no_jargon
  no_preamble
  return_only_summary

output_format: text
=== END_RUNPACK_TASK ===


# --- Expected result format ---

=== RUNPACK_RESULT ===
version: 1.1
legacy_aip_version: 1.1
aip_version: 1.1
packet_type: runpack_result
task_id: sum-001
status:  ok

[Summary appears here — 3 sentences maximum.]
=== END_RUNPACK_RESULT ===
